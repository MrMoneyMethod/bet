cmake_minimum_required(VERSION 3.20)
project(Edgecourt VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)

add_library(edgecourt_core src/core.cpp src/feeds.cpp src/history.cpp src/history_download.cpp)
target_include_directories(edgecourt_core PUBLIC src vendor)
target_link_libraries(edgecourt_core PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(edgecourt_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0603)
  target_link_libraries(edgecourt_core PUBLIC winhttp ws2_32 shell32)
endif()
if(MSVC)
  target_compile_options(edgecourt_core PUBLIC /utf-8 /W4 /EHsc)
  set_property(TARGET edgecourt_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(edgecourt_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

set(ASSET_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_assets.hpp")
add_custom_command(OUTPUT "${ASSET_HEADER}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_assets.py" --root "${CMAKE_CURRENT_SOURCE_DIR}" --output "${ASSET_HEADER}"
  DEPENDS scripts/embed_assets.py web/index.html web/styles.css web/app.js web/favicon.svg
  VERBATIM)
add_executable(Edgecourt src/main.cpp src/app.cpp "${ASSET_HEADER}")
target_include_directories(Edgecourt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(Edgecourt PRIVATE edgecourt_core)
if(MSVC)
  set_property(TARGET Edgecourt PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(MINGW)
  target_link_options(Edgecourt PRIVATE -static -static-libgcc -static-libstdc++)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(edgecourt_tests tests/test_core.cpp)
  target_link_libraries(edgecourt_tests PRIVATE edgecourt_core)
  if(MSVC)
    set_property(TARGET edgecourt_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME core_and_feeds COMMAND edgecourt_tests)
  add_test(NAME history_python COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_history.py")
  add_test(NAME history_native_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/history_native_smoke.py" "$<TARGET_FILE:Edgecourt>")
  add_test(NAME api_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/api_smoke.py" "$<TARGET_FILE:Edgecourt>")
endif()
install(TARGETS Edgecourt RUNTIME DESTINATION .)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
